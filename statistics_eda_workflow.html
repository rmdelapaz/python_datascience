Step 1: Initial Data Reconnaissance 🔎

Before diving deep, we need to understand what we're working with. This is like reading the case file before investigating.

class EDAWorkflow:
    """
    A systematic EDA workflow that can be applied to any dataset.
    Like having a seasoned data detective as your partner!
    """
    
    def __init__(self, df, target=None):
        self.df = df
        self.target = target
        self.report = {}
        
    def initial_reconnaissance(self):
        """Step 1: Get the lay of the land"""
        print("="*60)
        print("📊 INITIAL DATA RECONNAISSANCE")
        print("="*60)
        
        # Basic information
        print(f"\n📌 Dataset Shape: {self.df.shape[0]:,} rows × {self.df.shape[1]} columns")
        print(f"📌 Memory Usage: {self.df.memory_usage().sum() / 1024**2:.2f} MB")
        
        # Column types
        print("\n📋 Column Types:")
        type_counts = self.df.dtypes.value_counts()
        for dtype, count in type_counts.items():
            print(f"  • {dtype}: {count} columns")
        
        # Missing values overview
        missing = self.df.isnull().sum()
        missing_pct = 100 * missing / len(self.df)
        missing_df = pd.DataFrame({
            'Missing_Count': missing,
            'Missing_Percentage': missing_pct
        }).sort_values('Missing_Percentage', ascending=False)
        
        print("\n⚠️ Missing Values:")
        if missing_df['Missing_Count'].sum() > 0:
            print(missing_df[missing_df['Missing_Count'] > 0])
        else:
            print("  ✅ No missing values!")
        
        # Duplicates
        duplicates = self.df.duplicated().sum()
        print(f"\n🔄 Duplicate Rows: {duplicates} ({100*duplicates/len(self.df):.2f}%)")
        
        # Basic statistics
        print("\n📈 Quick Statistics:")
        print(self.df.describe(include='all').round(2))
        
        # Save to report
        self.report['shape'] = self.df.shape
        self.report['missing'] = missing_df
        self.report['duplicates'] = duplicates
        
        return self

Step 2: Univariate Analysis - Know Each Variable 📊

Like interviewing each witness separately, we need to understand each variable on its own before looking at relationships.

def univariate_analysis(df):
    """Analyze each variable independently"""
    print("="*60)
    print("📊 UNIVARIATE ANALYSIS")
    print("="*60)
    
    # Separate numerical and categorical columns
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    categorical_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    datetime_cols = df.select_dtypes(include=['datetime64']).columns.tolist()
    
    # Remove ID columns from analysis
    id_cols = [col for col in numerical_cols if 'id' in col.lower()]
    numerical_cols = [col for col in numerical_cols if col not in id_cols]
    
    print(f"\n📈 Numerical Variables ({len(numerical_cols)}): {numerical_cols}")
    print(f"📝 Categorical Variables ({len(categorical_cols)}): {categorical_cols}")
    print(f"📅 Datetime Variables ({len(datetime_cols)}): {datetime_cols}")
    
    # Analyze numerical variables
    if numerical_cols:
        fig, axes = plt.subplots(len(numerical_cols), 3, figsize=(15, 5*len(numerical_cols)))
        if len(numerical_cols) == 1:
            axes = axes.reshape(1, -1)
        
        for idx, col in enumerate(numerical_cols):
            data = df[col].dropna()
            
            # Distribution plot
            axes[idx, 0].hist(data, bins=50, edgecolor='black', alpha=0.7)
            axes[idx, 0].set_title(f'{col} - Distribution')
            axes[idx, 0].set_xlabel(col)
            axes[idx, 0].set_ylabel('Frequency')
            
            # Add mean and median lines
            axes[idx, 0].axvline(data.mean(), color='red', linestyle='--', label=f'Mean: {data.mean():.2f}')
            axes[idx, 0].axvline(data.median(), color='green', linestyle='--', label=f'Median: {data.median():.2f}')
            axes[idx, 0].legend()
            
            # Box plot
            axes[idx, 1].boxplot(data)
            axes[idx, 1].set_title(f'{col} - Box Plot')
            axes[idx, 1].set_ylabel(col)
            
            # Q-Q plot for normality check
            stats.probplot(data, dist="norm", plot=axes[idx, 2])
            axes[idx, 2].set_title(f'{col} - Q-Q Plot')
            
            # Statistical tests
            shapiro_stat, shapiro_p = stats.shapiro(data.sample(min(5000, len(data))))
            print(f"\n📊 {col}:")
            print(f"  • Mean: {data.mean():.2f}")
            print(f"  • Median: {data.median():.2f}")
            print(f"  • Std Dev: {data.std():.2f}")
            print(f"  • Skewness: {data.skew():.2f}")
            print(f"  • Kurtosis: {data.kurtosis():.2f}")
            print(f"  • Normality (Shapiro): p={shapiro_p:.4f} {'✅ Normal' if shapiro_p > 0.05 else '❌ Not Normal'}")
        
        plt.tight_layout()
        plt.show()
    
    # Analyze categorical variables
    if categorical_cols:
        fig, axes = plt.subplots(len(categorical_cols), 2, figsize=(12, 4*len(categorical_cols)))
        if len(categorical_cols) == 1:
            axes = axes.reshape(1, -1)
        
        for idx, col in enumerate(categorical_cols):
            # Value counts
            value_counts = df[col].value_counts()
            
            # Bar plot
            axes[idx, 0].bar(range(len(value_counts)), value_counts.values)
            axes[idx, 0].set_title(f'{col} - Frequency')
            axes[idx, 0].set_xlabel(col)
            axes[idx, 0].set_ylabel('Count')
            axes[idx, 0].set_xticks(range(len(value_counts)))
            axes[idx, 0].set_xticklabels(value_counts.index, rotation=45, ha='right')
            
            # Pie chart (if not too many categories)
            if len(value_counts) <= 10:
                axes[idx, 1].pie(value_counts.values, labels=value_counts.index, autopct='%1.1f%%')
                axes[idx, 1].set_title(f'{col} - Distribution')
            else:
                # Show top 10 for pie chart
                top_10 = value_counts.head(10)
                other_count = value_counts[10:].sum()
                if other_count > 0:
                    plot_values = top_10.tolist() + [other_count]
                    plot_labels = top_10.index.tolist() + ['Others']
                else:
                    plot_values = top_10.values
                    plot_labels = top_10.index
                axes[idx, 1].pie(plot_values, labels=plot_labels, autopct='%1.1f%%')
                axes[idx, 1].set_title(f'{col} - Top 10 + Others')
            
            print(f"\n📝 {col}:")
            print(f"  • Unique values: {df[col].nunique()}")
            print(f"  • Most common: {value_counts.index[0]} ({value_counts.iloc[0]:,} occurrences)")
            print(f"  • Least common: {value_counts.index[-1]} ({value_counts.iloc[-1]:,} occurrences)")
        
        plt.tight_layout()
        plt.show()

Step 3: Bivariate Analysis - Finding Relationships 🔗

Now we look for connections between pairs of variables. This is where patterns start to emerge!

def bivariate_analysis(df, target_col=None):
    """Analyze relationships between pairs of variables"""
    print("="*60)
    print("🔗 BIVARIATE ANALYSIS")
    print("="*60)
    
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    categorical_cols = df.select_dtypes(include=['object', 'category']).columns.tolist()
    
    # Remove ID columns
    numerical_cols = [col for col in numerical_cols if 'id' not in col.lower()]
    
    # 1. Numerical vs Numerical: Correlation Matrix
    if len(numerical_cols) > 1:
        print("\n📊 Numerical Correlations:")
        
        # Calculate correlations
        corr_matrix = df[numerical_cols].corr()
        
        # Plot heatmap
        plt.figure(figsize=(10, 8))
        sns.heatmap(corr_matrix, annot=True, cmap='coolwarm', center=0, 
                   square=True, linewidths=1, cbar_kws={"shrink": 0.8})
        plt.title('Correlation Matrix - Numerical Variables')
        plt.show()
        
        # Find strong correlations
        strong_corr = []
        for i in range(len(corr_matrix.columns)):
            for j in range(i+1, len(corr_matrix.columns)):
                if abs(corr_matrix.iloc[i, j]) > 0.5:
                    strong_corr.append({
                        'var1': corr_matrix.columns[i],
                        'var2': corr_matrix.columns[j],
                        'correlation': corr_matrix.iloc[i, j]
                    })
        
        if strong_corr:
            print("\n💪 Strong Correlations (|r| > 0.5):")
            for item in strong_corr:
                print(f"  • {item['var1']} ↔ {item['var2']}: {item['correlation']:.3f}")
    
    # 2. Categorical vs Numerical
    if categorical_cols and numerical_cols:
        print("\n📊 Categorical vs Numerical Relationships:")
        
        # Select most important combinations
        for cat_col in categorical_cols[:2]:  # Limit to first 2 categorical
            for num_col in numerical_cols[:2]:  # Limit to first 2 numerical
                if df[cat_col].nunique() <= 10:  # Only if reasonable number of categories
                    plt.figure(figsize=(10, 5))
                    
                    # Box plot
                    plt.subplot(1, 2, 1)
                    df.boxplot(column=num_col, by=cat_col, ax=plt.gca())
                    plt.title(f'{num_col} by {cat_col}')
                    plt.suptitle('')
                    
                    # Violin plot
                    plt.subplot(1, 2, 2)
                    cat_order = df[cat_col].value_counts().index
                    sns.violinplot(data=df, x=cat_col, y=num_col, order=cat_order)
                    plt.title(f'{num_col} Distribution by {cat_col}')
                    plt.xticks(rotation=45)
                    
                    plt.tight_layout()
                    plt.show()
                    
                    # Statistical test (ANOVA)
                    groups = [group[num_col].dropna() for name, group in df.groupby(cat_col)]
                    if len(groups) > 1:
                        f_stat, p_value = stats.f_oneway(*groups)
                        print(f"\n  {cat_col} vs {num_col}:")
                        print(f"    ANOVA F-statistic: {f_stat:.3f}")
                        print(f"    p-value: {p_value:.4f}")
                        if p_value < 0.05:
                            print(f"    ✅ Significant relationship!")
                        else:
                            print(f"    ❌ No significant relationship")
    
    # 3. Categorical vs Categorical
    if len(categorical_cols) > 1:
        print("\n📊 Categorical Associations:")
        
        for i, cat1 in enumerate(categorical_cols[:3]):
            for cat2 in categorical_cols[i+1:4]:
                if df[cat1].nunique() <= 10 and df[cat2].nunique() <= 10:
                    # Create crosstab
                    crosstab = pd.crosstab(df[cat1], df[cat2])
                    
                    # Chi-square test
                    chi2, p_value, dof, expected = stats.chi2_contingency(crosstab)
                    
                    # Cramér's V for effect size
                    n = crosstab.sum().sum()
                    min_dim = min(crosstab.shape[0] - 1, crosstab.shape[1] - 1)
                    cramers_v = np.sqrt(chi2 / (n * min_dim))
                    
                    print(f"\n  {cat1} vs {cat2}:")
                    print(f"    Chi-square: {chi2:.3f}")
                    print(f"    p-value: {p_value:.4f}")
                    print(f"    Cramér's V: {cramers_v:.3f}")
                    
                    if p_value < 0.05:
                        print(f"    ✅ Significant association!")
                        
                        # Visualize with heatmap
                        plt.figure(figsize=(8, 6))
                        sns.heatmap(crosstab, annot=True, fmt='d', cmap='YlOrRd')
                        plt.title(f'Crosstab: {cat1} vs {cat2}')
                        plt.show()

# Perform bivariate analysis
bivariate_analysis(df, target_col='amount')

Step 4: Outlier Detection - Finding the Unusual 🔍

Outliers can be errors, anomalies, or the most interesting insights in your data. Let's find them systematically!

def outlier_detection(df):
    """Systematic outlier detection using multiple methods"""
    print("="*60)
    print("🔍 OUTLIER DETECTION")
    print("="*60)
    
    numerical_cols = df.select_dtypes(include=['int64', 'float64']).columns.tolist()
    numerical_cols = [col for col in numerical_cols if 'id' not in col.lower()]
    
    outlier_report = {}
    
    for col in numerical_cols:
        data = df[col].dropna()
        outliers_dict = {}
        
        # Method 1: IQR (Interquartile Range)
        Q1 = data.quantile(0.25)
        Q3 = data.quantile(0.75)
        IQR = Q3 - Q1
        lower_bound = Q1 - 1.5 * IQR
        upper_bound = Q3 + 1.5 * IQR
        outliers_iqr = data[(data < lower_bound) | (data > upper_bound)]
        outliers_dict['IQR'] = len(outliers_iqr)
        
        # Method 2: Z-Score
        z_scores = np.abs(stats.zscore(data))
        outliers_zscore = data[z_scores > 3]
        outliers_dict['Z-Score'] = len(outliers_zscore)
        
        # Method 3: Modified Z-Score (MAD - Median Absolute Deviation)
        median = data.median()
        mad = np.median(np.abs(data - median))
        modified_z_scores = 0.6745 * (data - median) / (mad if mad else 1)
        outliers_mad = data[np.abs(modified_z_scores) > 3.5]
        outliers_dict['MAD'] = len(outliers_mad)
        
        # Method 4: Isolation Forest
        from sklearn.ensemble import IsolationForest
        iso_forest = IsolationForest(contamination=0.05, random_state=42)
        outliers_iso = iso_forest.fit_predict(data.values.reshape(-1, 1))
        outliers_dict['Isolation Forest'] = sum(outliers_iso == -1)
        
        outlier_report[col] = outliers_dict
        
        # Visualize outliers
        fig, axes = plt.subplots(1, 4, figsize=(16, 4))
        
        # Box plot
        axes[0].boxplot(data)
        axes[0].set_title(f'{col} - Box Plot')
        axes[0].set_ylabel('Value')
        
        # Histogram with outlier boundaries
        axes[1].hist(data, bins=50, alpha=0.7, edgecolor='black')
        axes[1].axvline(lower_bound, color='r', linestyle='--', label=f'IQR Lower: {lower_bound:.2f}')
        axes[1].axvline(upper_bound, color='r', linestyle='--', label=f'IQR Upper: {upper_bound:.2f}')
        axes[1].set_title(f'{col} - Distribution with IQR Bounds')
        axes[1].legend()
        
        # Z-score distribution
        axes[2].hist(z_scores, bins=50, alpha=0.7, edgecolor='black')
        axes[2].axvline(3, color='r', linestyle='--', label='Z = 3')
        axes[2].set_title(f'{col} - Z-Score Distribution')
        axes[2].legend()
        
        # Scatter plot with outliers highlighted
        axes[3].scatter(range(len(data)), data, alpha=0.5, s=10)
        outlier_indices = outliers_iqr.index
        axes[3].scatter(outlier_indices, outliers_iqr, color='red', s=20, label='Outliers')
        axes[3].set_title(f'{col} - Outliers Highlighted')
        axes[3].set_xlabel('Index')
        axes[3].set_ylabel('Value')
        axes[3].legend()
        
        plt.tight_layout()
        plt.show()
        
        print(f"\n📊 {col}:")
        print(f"  • Total observations: {len(data)}")
        for method, count in outliers_dict.items():
            pct = 100 * count / len(data)
            print(f"  • {method}: {count} outliers ({pct:.2f}%)")
    
    # Summary table
    outlier_df = pd.DataFrame(outlier_report).T
    print("\n📋 Outlier Summary Table:")
    print(outlier_df)
    
    return outlier_df

def pattern_discovery(df):
    """Discover patterns and generate insights"""
    print("="*60)
    print("💡 PATTERN DISCOVERY & INSIGHTS")
    print("="*60)
    
    insights = []
    
    # Temporal patterns
    if 'date' in df.columns:
        df['month'] = pd.to_datetime(df['date']).dt.month
        df['day_of_week'] = pd.to_datetime(df['date']).dt.dayofweek
        
        # Monthly trends
        monthly_stats = df.groupby('month')['amount'].agg(['mean', 'sum', 'count'])
        
        plt.figure(figsize=(15, 5))
        
        plt.subplot(1, 3, 1)
        monthly_stats['sum'].plot(kind='bar')
        plt.title('Monthly Revenue')
        plt.xlabel('Month')
        plt.ylabel('Total Revenue')
        
        plt.subplot(1, 3, 2)
        monthly_stats['count'].plot(kind='bar')
        plt.title('Monthly Transaction Count')
        plt.xlabel('Month')
        plt.ylabel('Number of Transactions')
        
        plt.subplot(1, 3, 3)
        monthly_stats['mean'].plot(kind='bar')
        plt.title('Average Transaction Value by Month')
        plt.xlabel('Month')
        plt.ylabel('Average Amount')
        
        plt.tight_layout()
        plt.show()
        
        # Best and worst months
        best_month = monthly_stats['sum'].idxmax()
        worst_month = monthly_stats['sum'].idxmin()
        insights.append(f"📈 Best revenue month: {best_month} (${monthly_stats.loc[best_month, 'sum']:,.2f})")
        insights.append(f"📉 Worst revenue month: {worst_month} (${monthly_stats.loc[worst_month, 'sum']:,.2f})")
    
    # Customer segmentation patterns
    if 'customer_id' in df.columns:
        customer_stats = df.groupby('customer_id').agg({
            'amount': ['sum', 'mean', 'count'],
            'product_category': lambda x: x.mode()[0] if not x.empty else 'Unknown'
        })
        customer_stats.columns = ['total_spent', 'avg_transaction', 'transaction_count', 'favorite_category']
        
        # RFM-like segmentation
        customer_stats['value_segment'] = pd.qcut(customer_stats['total_spent'], 
                                                   q=3, labels=['Low', 'Medium', 'High'])
        customer_stats['frequency_segment'] = pd.qcut(customer_stats['transaction_count'], 
                                                      q=3, labels=['Low', 'Medium', 'High'])
        
        # Segment distribution
        segment_dist = customer_stats.groupby(['value_segment', 'frequency_segment']).size().unstack(fill_value=0)
        
        plt.figure(figsize=(8, 6))
        sns.heatmap(segment_dist, annot=True, fmt='d', cmap='YlOrRd')
        plt.title('Customer Segmentation Matrix')
        plt.xlabel('Frequency Segment')
        plt.ylabel('Value Segment')
        plt.show()
        
        # High-value customers
        high_value_customers = customer_stats[customer_stats['value_segment'] == 'High']
        insights.append(f"👑 High-value customers: {len(high_value_customers)} ({100*len(high_value_customers)/len(customer_stats):.1f}%)")
        insights.append(f"💰 They contribute ${high_value_customers['total_spent'].sum():,.2f} in revenue")
    
    # Product patterns
    if 'product_category' in df.columns:
        product_stats = df.groupby('product_category')['amount'].agg(['sum', 'mean', 'count'])
        
        fig, axes = plt.subplots(1, 2, figsize=(12, 5))
        
        # Revenue by category
        product_stats['sum'].sort_values().plot(kind='barh', ax=axes[0])
        axes[0].set_title('Revenue by Product Category')
        axes[0].set_xlabel('Total Revenue')
        
        # Average transaction by category
        product_stats['mean'].sort_values().plot(kind='barh', ax=axes[1])
        axes[1].set_title('Average Transaction Value by Category')
        axes[1].set_xlabel('Average Amount')
        
        plt.tight_layout()
        plt.show()
        
        best_category = product_stats['sum'].idxmax()
        insights.append(f"🏆 Top revenue category: {best_category} (${product_stats.loc[best_category, 'sum']:,.2f})")
    
    # Payment method patterns
    if 'payment_method' in df.columns:
        payment_stats = df.groupby('payment_method').agg({
            'amount': ['mean', 'sum', 'count']
        })
        payment_stats.columns = ['avg_amount', 'total_amount', 'count']
        
        insights.append(f"💳 Most popular payment: {payment_stats['count'].idxmax()} ({payment_stats['count'].max()} transactions)")
        insights.append(f"💎 Highest value payment method: {payment_stats['avg_amount'].idxmax()} (avg ${payment_stats['avg_amount'].max():.2f})")
    
    # Display insights
    print("\n🎯 KEY INSIGHTS DISCOVERED:")
    print("-" * 40)
    for i, insight in enumerate(insights, 1):
        print(f"{i}. {insight}")
    
    return insights

# Advanced technique 2: Correlation network visualization
def plot_correlation_network(df, threshold=0.3):
    """Create network graph of correlations"""
    import networkx as nx
    
    numerical_cols = df.select_dtypes(include=['float64', 'int64']).columns
    numerical_cols = [col for col in numerical_cols if 'id' not in col.lower()]
    
    if len(numerical_cols) < 2:
        return
    
    corr = df[numerical_cols].corr()
    
    # Create network
    G = nx.Graph()
    
    # Add nodes
    for col in numerical_cols:
        G.add_node(col)
    
    # Add edges for correlations above threshold
    for i in range(len(numerical_cols)):
        for j in range(i+1, len(numerical_cols)):
            if abs(corr.iloc[i, j]) > threshold:
                G.add_edge(numerical_cols[i], numerical_cols[j], 
                          weight=abs(corr.iloc[i, j]),
                          correlation=corr.iloc[i, j])
    
    # Plot
    plt.figure(figsize=(10, 8))
    pos = nx.spring_layout(G)
    
    # Draw nodes
    nx.draw_networkx_nodes(G, pos, node_size=1000, node_color='lightblue')
    
    # Draw edges with varying thickness
    edges = G.edges()
    weights = [G[u][v]['weight'] * 3 for u, v in edges]
    colors = ['red' if G[u][v]['correlation'] < 0 else 'green' for u, v in edges]
    nx.draw_networkx_edges(G, pos, width=weights, edge_color=colors, alpha=0.6)
    
    # Draw labels
    nx.draw_networkx_labels(G, pos)
    
    plt.title(f'Correlation Network (|r| > {threshold})')
    plt.axis('off')
    plt.show()

plot_correlation_network(df, threshold=0.3)

from sklearn.preprocessing import StandardScaler
from sklearn.decomposition import PCA

def perform_pca_analysis(df):
    """PCA for pattern discovery in high-dimensional data"""
    
    # Select numerical features
    numerical_cols = df.select_dtypes(include=['float64', 'int64']).columns
    numerical_cols = [col for col in numerical_cols if 'id' not in col.lower()]
    
    if len(numerical_cols) < 3:
        print("Not enough numerical features for PCA")
        return
    
    # Prepare data
    X = df[numerical_cols].dropna()
    X_scaled = StandardScaler().fit_transform(X)
    
    # Perform PCA
    pca = PCA()
    X_pca = pca.fit_transform(X_scaled)
    
    # Explained variance
    exp_var_ratio = pca.explained_variance_ratio_
    cum_var_ratio = np.cumsum(exp_var_ratio)
    
    # Plot
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    # Scree plot
    axes[0].bar(range(1, len(exp_var_ratio)+1), exp_var_ratio)
    axes[0].plot(range(1, len(exp_var_ratio)+1), cum_var_ratio, 'r-', marker='o')
    axes[0].set_xlabel('Principal Component')
    axes[0].set_ylabel('Explained Variance Ratio')
    axes[0].set_title('PCA Scree Plot')
    axes[0].axhline(y=0.9, color='g', linestyle='--', label='90% variance')
    axes[0].legend()
    
    # Biplot (first 2 components)
    axes[1].scatter(X_pca[:, 0], X_pca[:, 1], alpha=0.5)
    axes[1].set_xlabel(f'PC1 ({exp_var_ratio[0]:.2%} variance)')
    axes[1].set_ylabel(f'PC2 ({exp_var_ratio[1]:.2%} variance)')
    axes[1].set_title('PCA Biplot - First 2 Components')
    
    # Add feature vectors
    for i, col in enumerate(numerical_cols):
        axes[1].arrow(0, 0, pca.components_[0, i]*3, pca.components_[1, i]*3,
                     head_width=0.1, head_length=0.1, fc='red', ec='red')
        axes[1].text(pca.components_[0, i]*3.5, pca.components_[1, i]*3.5, col,
                    fontsize=9, ha='center')
    
    plt.tight_layout()
    plt.show()
    
    print(f"✅ Components needed for 90% variance: {np.argmax(cum_var_ratio >= 0.9) + 1}")

# Common EDA mistakes and how to avoid them

def eda_pitfalls_demo():
    """Demonstrate common EDA pitfalls"""
    print("="*60)
    print("⚠️ COMMON EDA PITFALLS")
    print("="*60)
    
    # Pitfall 1: Simpson's Paradox
    print("\n1️⃣ Simpson's Paradox - Aggregation can reverse trends!")
    
    # Create example data
    np.random.seed(42)
    data_simpson = pd.DataFrame({
        'department': np.repeat(['A', 'B'], 100),
        'experience': np.concatenate([np.random.normal(5, 2, 100), np.random.normal(10, 2, 100)]),
        'salary': np.concatenate([np.random.normal(50000, 5000, 100), np.random.normal(70000, 5000, 100)])
    })
    
    # Add negative correlation within groups but positive overall
    data_simpson.loc[data_simpson['department'] == 'A', 'salary'] -= data_simpson.loc[data_simpson['department'] == 'A', 'experience'] * 1000
    data_simpson.loc[data_simpson['department'] == 'B', 'salary'] -= data_simpson.loc[data_simpson['department'] == 'B', 'experience'] * 1000
    
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    # Overall correlation
    axes[0].scatter(data_simpson['experience'], data_simpson['salary'], alpha=0.5)
    axes[0].set_xlabel('Experience')
    axes[0].set_ylabel('Salary')
    axes[0].set_title('Overall: Positive Correlation')
    z = np.polyfit(data_simpson['experience'], data_simpson['salary'], 1)
    p = np.poly1d(z)
    axes[0].plot(data_simpson['experience'], p(data_simpson['experience']), "r--")
    
    # By department
    for dept in ['A', 'B']:
        dept_data = data_simpson[data_simpson['department'] == dept]
        axes[1].scatter(dept_data['experience'], dept_data['salary'], label=f'Dept {dept}', alpha=0.5)
        z = np.polyfit(dept_data['experience'], dept_data['salary'], 1)
        p = np.poly1d(z)
        axes[1].plot(dept_data['experience'], p(dept_data['experience']), "--")
    
    axes[1].set_xlabel('Experience')
    axes[1].set_ylabel('Salary')
    axes[1].set_title('By Department: Negative Correlations!')
    axes[1].legend()
    
    plt.tight_layout()
    plt.show()
    
    print("⚠️ Lesson: Always check subgroups - aggregated data can be misleading!")
    
    # Pitfall 2: Ignoring time dependencies
    print("\n2️⃣ Ignoring Time Dependencies")
    
    # Create time series with trend
    dates = pd.date_range('2023-01-01', periods=365, freq='D')
    trend = np.linspace(100, 200, 365)
    seasonal = 20 * np.sin(np.arange(365) * 2 * np.pi / 30)  # Monthly pattern
    noise = np.random.normal(0, 10, 365)
    values = trend + seasonal + noise
    
    ts_data = pd.DataFrame({'date': dates, 'value': values})
    
    fig, axes = plt.subplots(1, 2, figsize=(12, 5))
    
    # Without considering time
    axes[0].hist(ts_data['value'], bins=30, edgecolor='black')
    axes[0].set_title('Distribution (ignoring time) - Looks normal!')
    axes[0].set_xlabel('Value')
    
    # With time
    axes[1].plot(ts_data['date'], ts_data['value'])
    axes[1].set_title('Time Series - Clear trend and seasonality!')
    axes[1].set_xlabel('Date')
    axes[1].set_ylabel('Value')
    
    plt.tight_layout()
    plt.show()
    
    print("⚠️ Lesson: Always check if your data has time dependencies!")
    
    # Pitfall 3: Correlation vs Causation
    print("\n3️⃣ Correlation ≠ Causation")
    
    # Spurious correlation example
    years = np.arange(2010, 2020)
    ice_cream_sales = np.array([100, 105, 108, 112, 118, 125, 132, 138, 145, 152])
    shark_attacks = np.array([45, 48, 47, 52, 55, 58, 62, 65, 68, 70])
    
    correlation = np.corrcoef(ice_cream_sales, shark_attacks)[0, 1]
    
    plt.figure(figsize=(10, 5))
    plt.scatter(ice_cream_sales, shark_attacks)
    z = np.polyfit(ice_cream_sales, shark_attacks, 1)
    p = np.poly1d(z)
    plt.plot(ice_cream_sales, p(ice_cream_sales), "r--")
    plt.xlabel('Ice Cream Sales (thousands)')
    plt.ylabel('Shark Attacks')
    plt.title(f'Spurious Correlation (r = {correlation:.3f}) - Both driven by summer/temperature!')
    plt.show()
    
    print("⚠️ Lesson: Correlation doesn't imply causation - look for confounding variables!")

def generate_eda_report(df, output_file='eda_report.txt'):
    """Generate a comprehensive EDA report"""
    
    with open(output_file, 'w') as f:
        f.write("="*60 + "\n")
        f.write("EXPLORATORY DATA ANALYSIS REPORT\n")
        f.write("="*60 + "\n\n")
        
        # Executive Summary
        f.write("EXECUTIVE SUMMARY\n")
        f.write("-"*40 + "\n")
        f.write(f"Dataset: {df.shape[0]:,} rows × {df.shape[1]} columns\n")
        f.write(f"Date Range: {df['date'].min()} to {df['date'].max()}\n" if 'date' in df.columns else "")
        f.write(f"Missing Values: {df.isnull().sum().sum()} total\n")
        f.write(f"Memory Usage: {df.memory_usage().sum() / 1024**2:.2f} MB\n\n")
        
        # Key Findings
        f.write("KEY FINDINGS\n")
        f.write("-"*40 + "\n")
        
        numerical_cols = df.select_dtypes(include=['float64', 'int64']).columns
        for col in numerical_cols[:5]:  # Top 5 numerical columns
            if 'id' not in col.lower():
                f.write(f"\n{col}:\n")
                f.write(f"  • Mean: {df[col].mean():.2f}\n")
                f.write(f"  • Median: {df[col].median():.2f}\n")
                f.write(f"  • Std Dev: {df[col].std():.2f}\n")
                f.write(f"  • Range: [{df[col].min():.2f}, {df[col].max():.2f}]\n")
        
        # Data Quality Issues
        f.write("\nDATA QUALITY ISSUES\n")
        f.write("-"*40 + "\n")
        
        missing = df.isnull().sum()
        for col in missing[missing > 0].index:
            f.write(f"  • {col}: {missing[col]} missing ({100*missing[col]/len(df):.1f}%)\n")
        
        # Recommendations
        f.write("\nRECOMMENDATIONS\n")
        f.write("-"*40 + "\n")
        f.write("1. Handle missing values in categorical features\n")
        f.write("2. Investigate outliers in amount column\n")
        f.write("3. Consider feature engineering for date components\n")
        f.write("4. Segment customers based on behavior patterns\n")
        
        f.write("\n" + "="*60 + "\n")
        f.write("Report generated successfully!\n")
    
    print(f"✅ EDA Report saved to {output_file}")
    return output_file

❓ Common Questions at This Stage

How much EDA is "enough" before I start modeling?

Enough to trust your data: you understand its shape, have addressed missing values and duplicates, know the main relationships, and have investigated notable outliers. EDA is iterative — you'll return to it as questions arise.

An outlier detector flagged some points — should I remove them?

Not automatically. Investigate first: outliers can be data-entry errors (fix or drop) or the most interesting signal in the data (keep and study). Use domain knowledge, not just the algorithm.

Why do different outlier methods disagree?

Each makes different assumptions — IQR and z-score assume roughly symmetric data, MAD is robust to skew, and Isolation Forest is model-based. Comparing them gives a fuller picture than trusting any single one.
